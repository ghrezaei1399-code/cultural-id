'badge-rejected' : 'badge-pending'; const statusText = user.status === 'approved' ? 'تأیید شده' : user.status === 'rejected' ? 'رد شده' : 'در انتظار'; const valuesText = user.values ? user.values.join(', ') : '---'; const optionalCode = user.optionalCode || '---'; const email = user.communicationEmail || '---'; const date = user.registrationDate ? new Date(user.registrationDate).toLocaleDateString('fa-IR') : '---'; const allowConnection = user.allowConnection ?
