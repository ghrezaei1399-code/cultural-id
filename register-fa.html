export default async function handler(req, res) { if (req.method !== 'POST') return res.status(405).json({ error: 'Method not allowed' }); const token = process.env.GH_TOKEN; if (!token) return res.status(500).json({ error: 'Server config error' }); try { const { values, priorities, optionalCode, communicationEmail } = req.body; // اعتبارسنجی ساده if (!values || !Array.isArray(values)) { return res.status(400).json({ error: 'Invalid data format' }); } const owner = 'ghrezaei1399-code'; const repo = 'cultural-id'; const cardCode = `CIM-${Date.now().toString().slice(-6)}-${Math.floor(Math.random() * 10000).toString().padStart(4, '0')}`; const userData = { cardCode, displayCode: optionalCode ? `${cardCode} - ${optionalCode}` : cardCode, values, priorities: priorities || [], communicationEmail: communicationEmail || null, registrationDate: new Date().toISOString(), status: 'pending', rank: 0 }; // ذخیره کاربر await fetch(`https://api.github.com/repos/${owner}/${repo}/contents/data/active/${cardCode}.json`, { method: 'PUT', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ message: `New user: ${cardCode}`, content: Buffer.from(JSON.stringify(userData)).toString('base64'), branch: 'main' }) }); // آپدیت ایندکس (با مدیریت خطای SHA) let sha = null; let indexData = []; const idxRes = await fetch(`https://api.github.com/repos/${owner}/${repo}/contents/data/index.json`, { headers: { 'Authorization': `Bearer ${token}` } }); if (idxRes.ok) { const idxFile = await idxRes.json(); sha = idxFile.sha; indexData = JSON.parse(Buffer.from(idxFile.content, 'base64').toString()); } indexData.push({ cardCode, displayCode: userData.displayCode, country: 'Unknown', rank: 0, status: 'pending', registrationDate: userData.registrationDate }); await fetch(`https://api.github.com/repos/${owner}/${repo}/contents/data/index.json`, { method: 'PUT', headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ message: `Update index: ${cardCode}`, content: Buffer.from(JSON.stringify(indexData)).toString('base64'), sha: sha, branch: 'main' }) }); return res.status(200).json({ success: true, cardCode }); } catch (error) { console.error(error); return res.status(500).json({ error: error.message }); } }
