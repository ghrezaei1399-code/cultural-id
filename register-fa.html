export default async function handler(req, res) { if (req.method !== 'POST') { return res.status(405).json({ error: 'Method not allowed' }); } const token = process.env.GH_TOKEN; if (!token) { console.error('GH_TOKEN is missing in environment variables'); return res.status(500).json({ error: 'Server configuration error: Missing GH_TOKEN' }); } const { values, priorities, optionalCode, communicationEmail } = req.body; if (!values || !Array.isArray(values) || values.length !== 7) { return res.status(400).json({ error: 'لطفاً تمام ۷ ارزش فرهنگی را وارد کنید.' }); } const owner = 'ghrezaei1399-code'; const repo = 'cultural-id'; try { // تولید کد کارت یکتا const timestamp = Date.now().toString().slice(-6); const randomPart = Math.floor(Math.random() * 10000).toString().padStart(4, '0'); const cardCode = `CIM-${timestamp}-${randomPart}`; // ساخت داده‌های کاربر const userData = { cardCode, displayCode: optionalCode ? `${cardCode} - ${optionalCode}` : cardCode, values, priorities: priorities || [1, 2, 3, 4, 5, 6, 7], // اولویت‌های پیش‌فرض اگر ارسال نشده باشد communicationEmail: communicationEmail || null, registrationDate: new Date().toISOString(), status: 'pending', // نیاز به تایید ادمین rank: 0 }; // ۱. ذخیره فایل کاربر در data/active/ const userPath = `data/active/${cardCode}.json`; const userContent = Buffer.from(JSON.stringify(userData, null, 2), 'utf8').toString('base64'); await fetch(`https://api.github.com/repos/${owner}/${repo}/contents/${userPath}`, { method: 'PUT', headers: { 'Authorization': `Bearer ${token}`, 'Accept': 'application/vnd.github.v3+json', 'Content-Type': 'application/json' }, body: JSON.stringify({ message: `New registration: ${cardCode}`, content: userContent, branch: 'main' }) }); // ۲. آپدیت کردن index.json const indexPath = 'data/index.json'; // خواندن index فعلی const indexRes = await fetch(`https://api.github.com/repos/${owner}/${repo}/contents/${indexPath}`, { headers: { 'Authorization': `Bearer ${token}`, 'Accept': 'application/vnd.github.v3+json' } }); let indexData = []; let sha = null; if (indexRes.ok) { const indexFile = await indexRes.json(); sha = indexFile.sha; const jsonString = Buffer.from(indexFile.content, 'base64').toString('utf8'); indexData = JSON.parse(jsonString); } // افزودن کاربر جدید به لیست indexData.push({ cardCode, displayCode: userData.displayCode, country: 'Unknown', // بعداً توسط کاربر یا ادمین تکمیل می‌شود rank: 0, status: 'pending', registrationDate: userData.registrationDate }); // ذخیره index.json جدید const newIndexContent = Buffer.from(JSON.stringify(indexData, null, 2), 'utf8').toString('base64'); await fetch(`https://api.github.com/repos/${owner}/${repo}/contents/${indexPath}`, { method: 'PUT', headers: { 'Authorization': `Bearer ${token}`, 'Accept': 'application/vnd.github.v3+json', 'Content-Type': 'application/json' }, body: JSON.stringify({ message: `Add user to index: ${cardCode}`, content: newIndexContent, sha: sha, // SHA ضروری است برای آپدیت فایل موجود branch: 'main' }) }); return res.status(200).json({ success: true, cardCode, message: 'ثبت‌نام با موفقیت انجام شد. کارت شما پس از تایید ادمین فعال خواهد شد.' }); } catch (error) { console.error('Registration Error:', error); return res.status(500).json({ error: 'خطا در ثبت‌نام: ' + error.message }); } }
